shared var rx: Ring = Ring.(head: 0, tail: 0)
pub shared var ticks: u64 = 0

@interrupt
fn uart_isr() uses inout rx {
    lock rx {
        push(inout rx, read_data_register())
    }
}

fn next_byte() -> ?u8 uses inout rx, ticks {
    lock rx, ticks {
        if ticks == 0 {
            return none
        }
        return pop(inout rx)
    }
}

fn read_remote() uses inout net.queue {
    lock net.queue {
        clear(inout net.queue)
    }
}
